//iChannel0
void mainImage(out vec4 fragColor, in vec2 fragCoord){
vec2 uv=fragCoord.xy/iResolution.xy;
vec2 px=1.0/iResolution.xy;
vec2 vel=texture2D(iChannel0,uv).xy;
vec2 prevUV=uv-vel*0.01;
vec2 advected=texture2D(iChannel0,prevUV).xy;
vec2 mouse=iMouse.xy/iResolution.xy;
float aspect=iResolution.x/iResolution.y;
vec2 muv=vec2(mouse.x*aspect,mouse.y);
vec2 uva=vec2(uv.x*aspect,uv.y);
float d=length(uva-muv);
vec2 force=normalize(uva-muv+0.0001)*smoothstep(0.2,0.0,d)*0.5;
if(iMouse.z>0.0)advected+=force;
advected*=0.99;
fragColor=vec4(advected,0.0,1.0);
}

//iChannel1
void mainImage(out vec4 fragColor, in vec2 fragCoord){
vec2 uv=fragCoord.xy/iResolution.xy;
vec2 vel=texture2D(iChannel0,uv).xy;
vec2 prevUV=uv-vel*0.01;
vec4 density=texture2D(iChannel1,prevUV);
density.rgb*=0.995;
vec2 mouse=iMouse.xy/iResolution.xy;
float d=distance(uv,mouse);
float inject=smoothstep(0.1,0.0,d);
vec3 col=vec3(1.0,0.5,0.1)*inject;
if(iMouse.z>0.0)density.rgb+=col*0.2;
fragColor=density;
}

//Main
void mainImage(out vec4 fragColor, in vec2 fragCoord){
vec2 uv=fragCoord.xy/iResolution.xy;
vec4 d=texture2D(iChannel1,uv);
fragColor=vec4(d.rgb,1.0);
}
